  98: class AffinityHeadsTransformer(nn.Module):
  99:     def __init__(
 100:         self,
 101:         token_z: int,
 102:         hidden_dim: int,
 103:         **kwargs,
 104:     ):
 105:         super().__init__()
 106:         self.affinity_out_mlp = nn.Sequential(
 107:             nn.Linear(token_z, token_z),
 108:             nn.ReLU(),
 109:             nn.Linear(token_z, hidden_dim),
 110:             nn.ReLU(),
 111:         )
 112:         self.to_affinity_pred_value = nn.Sequential(
 113:             nn.Linear(hidden_dim, hidden_dim),
 114:             nn.ReLU(),
 115:             nn.Linear(hidden_dim, hidden_dim),
 116:             nn.ReLU(),
 117:             nn.Linear(hidden_dim, 1),
 118:         )
 119:         self.to_affinity_pred_score = nn.Sequential(
 120:             nn.Linear(hidden_dim, hidden_dim),
 121:             nn.ReLU(),
 122:             nn.Linear(hidden_dim, hidden_dim),
 123:             nn.ReLU(),
 124:             nn.Linear(hidden_dim, 1),
 125:         )
 126:         self.to_affinity_logits_binary = nn.Linear(1, 1)
 127: 
 128:     def forward(
 129:         self,
 130:         z: torch.Tensor,
 131:         feats: dict[str, torch.Tensor],
 132:     ) -> dict[str, torch.Tensor]:
 133:         pad_token_mask = feats["token_pad_mask"].unsqueeze(-1)
 134:         rec_mask = (feats["mol_type"] == 0).float().unsqueeze(-1) * pad_token_mask
 135:         lig_mask = feats["affinity_token_mask"].float().unsqueeze(-1) * pad_token_mask
 136: 
 137:         cross_pair_mask = (
 138:             lig_mask[:, :, None] * rec_mask[:, None, :]
 139:             + rec_mask[:, :, None] * lig_mask[:, None, :]
 140:             + (lig_mask[:, :, None] * lig_mask[:, None, :])
 141:         ) * (
 142:             1
 143:             - torch.eye(lig_mask.shape[1], device=lig_mask.device)
 144:             .unsqueeze(-1)
 145:             .unsqueeze(0)
 146:         )
 147: 
 148:         g = torch.sum(z * cross_pair_mask, dim=(1, 2)) / (
 149:             torch.sum(cross_pair_mask, dim=(1, 2)) + 1e-7
 150:         )
 151:         g = self.affinity_out_mlp(g)
 152:         affinity_pred_value = self.to_affinity_pred_value(g).reshape(-1, 1)
 153:         out_dict = {"affinity_pred_value": affinity_pred_value, "affinity_repr": g}
 154: 
 155:         affinity_pred_score = self.to_affinity_pred_score(g).reshape(-1, 1)
 156:         out_dict["affinity_logits_binary"] = self.to_affinity_logits_binary(
 157:             affinity_pred_score
 158:         ).reshape(-1, 1)
 159: 
 160:         return out_dict

  19:         dropout: float = 0.0,
  20:         **kwargs,  # for unused kwargs
  21:     ) -> None:
  22:         super().__init__()
  23:         self.use_esm_all_layers = use_esm_all_layers
  24:         self.esm_num_layers = esm_num_layers
  25:         if use_esm_all_layers:
  26:             self.esm_layer_weights = nn.Parameter(torch.zeros(esm_num_layers))
  27: 
  28:         self.esm_mlp = nn.Sequential(
  29:             nn.LayerNorm(esm_embed_dim),
  30:             nn.Linear(esm_embed_dim, token_s),
  31:             nn.ReLU(),
  32:             nn.Dropout(dropout),
  33:             nn.Linear(token_s, token_s),
  34:         )
  35:         self.s_inputs_proj = nn.Linear(token_s, token_s)
  36: 
  37:         # Single-to-pair (outer product in token_z)
  38:         self.esm_z_1 = nn.Linear(token_s, token_z, bias=False)
  39:         self.esm_z_2 = nn.Linear(token_s, token_z, bias=False)
  40:         initialize.gating_init_(self.esm_z_1.weight)
  41:         initialize.gating_init_(self.esm_z_2.weight)
  42:         initialize.gating_init_(self.s_inputs_proj.weight)
  43: 
  44:     def forward(
  45:         self,
  46:         z: Tensor,
  47:         s_inputs: Tensor,
  48:         s_esm: Tensor,
  49:         pair_mask: Tensor,
  50:         use_kernels: bool = False,
  51:     ) -> Tensor:
  52:         """Update z using s_inputs and s_esm."""
  53:         if s_esm.dim() == 4 and self.use_esm_all_layers:
  54:             weights = self.esm_layer_weights.softmax(0)
  55:             s_esm = torch.einsum("bnld,l->bnd", s_esm, weights)
  56: 
  57:         s_esm_proj = self.esm_mlp(s_esm)
  58:         s = self.s_inputs_proj(s_inputs) + s_esm_proj
  59: 
  60:         left = self.esm_z_1(s)  # [B, N, token_z]
  61:         right = self.esm_z_2(s)  # [B, N, token_z]
  62:         delta_z = left[:, :, None, :] + right[:, None, :, :]  # [B, N, N, token_z]
  63:         delta_z = delta_z * pair_mask.unsqueeze(
  64:             -1
  65:         )  # [B, N, N, token_z] (mask out invalid pairs)
  66:         z = z + delta_z  # update pairwise representation with the delta_z
  67: 
  68:         return z

  54:         x: torch.Tensor

  56:

  60:             The output data of shape (..., D)

 204:         x_in = x

 269:             self.linear_g = Linear(

 275:     def _prep_qkv(
 276:         self, q_x: torch.Tensor, kv_x: torch.Tensor, apply_scale: bool = True
 277:     ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
 278:         # [*, Q/K/V, H * C_hidden]
 279:         q = self.linear_q(q_x)
 280:         k = self.linear_k(kv_x)
 281:         v = self.linear_v(kv_x)
 282: 
 283:         # [*, Q/K, H, C_hidden]
 284:         q = q.view(q.shape[:-1] + (self.no_heads, -1))
 285:         k = k.view(k.shape[:-1] + (self.no_heads, -1))
 286:         v = v.view(v.shape[:-1] + (self.no_heads, -1))
 287: 
 288:         # [*, H, Q/K, C_hidden]
 289:         q = q.transpose(-2, -3)
 290:         k = k.transpose(-2, -3)
 291:         v = v.transpose(-2, -3)
 292: 
 293:         if apply_scale:
 294:             q /= math.sqrt(self.c_hidden)
 295: 
 296:         return q, k, v

 298:     def _wrap_up(self, o: torch.Tensor, q_x: torch.Tensor) -> torch.Tensor:
 299:         if self.linear_g is not None:
 300:             g = self.sigmoid(self.linear_g(q_x))
 301: 
 302:             # [*, Q, H, C_hidden]
 303:             g = g.view(g.shape[:-1] + (self.no_heads, -1))
 304:             o = o * g
 305: 
 306:         # [*, Q, H * C_hidden]
 307:         o = flatten_final_dims(o, 2)
 308: 
 309:         # [*, Q, C_q]
 310:         o = self.linear_o(o)
 311: 
 312:         return o
